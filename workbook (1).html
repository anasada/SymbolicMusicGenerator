class MusicRNN(nn.Module):
    def __init__(self, vocab_size, embedding_dim, hidden_dim, num_layers):
        super(MusicRNN, self).__init__()
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.rnn = nn.LSTM(
            input_size=embedding_dim,
            hidden_size=hidden_dim,
            num_layers=num_layers,
            batch_first=True
        )
        self.fc = nn.Linear(hidden_dim, vocab_size)

    def forward(self, x, hidden=None):
        # x: (batch_size, seq_length)
        x = self.embedding(x)  # (batch_size, seq_length, embedding_dim)
        out, hidden = self.rnn(x, hidden)  # out: (batch_size, seq_length, hidden_dim)
        out = self.fc(out)  # (batch_size, seq_length, vocab_size)
        return out, hidden

def train(model, train_loader, val_loader, vocab_size, num_epochs=10, lr=0.001, device='cuda'):
    model = model.to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)

    for epoch in range(num_epochs):
        # --------- Training ---------
        model.train()
        total_train_loss = 0

        batch_num = 0
        for batch in train_loader:
            batch_num += 1
            if batch_num % 100 == 0:
              print(f"Training Batch {batch_num}/{len(train_loader)}")
            batch = batch['input_ids'].to(device)  # (batch_size, seq_length)

            inputs = batch[:, :-1]
            targets = batch[:, 1:]

            optimizer.zero_grad()
            outputs, _ = model(inputs)
            outputs = outputs.reshape(-1, vocab_size)
            targets = targets.reshape(-1)

            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()

            total_train_loss += loss.item()

        avg_train_loss = total_train_loss / len(train_loader)

        # --------- Validation ---------
        model.eval()
        total_val_loss = 0
        with torch.no_grad():
            batch_num = 0
            for batch in val_loader:
                batch_num += 1
                if batch_num % 100 == 0:
                  print(f"Testing Batch {batch_num}/{len(val_loader)}")
                batch = batch['input_ids'].to(device)

                inputs = batch[:, :-1]
                targets = batch[:, 1:]

                outputs, _ = model(inputs)
                outputs = outputs.reshape(-1, vocab_size)
                targets = targets.reshape(-1)

                loss = criterion(outputs, targets)
                total_val_loss += loss.item()

        avg_val_loss = total_val_loss / len(val_loader)

        print(f"Epoch {epoch+1}/{num_epochs} | Train Loss: {avg_train_loss:.4f} | Val Loss: {avg_val_loss:.4f}")

    torch.save(model.state_dict(), "symbolicUncond_model.pth")

def sample(model, start_token, max_length=100, temperature=1.0, device='cuda'):
    model = model.to(device)
    model.eval()

    generated = [start_token]
    input_token = torch.tensor([[start_token]], device=device)  # (1, 1)

    hidden = None

    for _ in range(max_length):
        output, hidden = model(input_token, hidden)  # output: (1, 1, vocab_size)
        output = output[:, -1, :]  # take the last output
        output = output / temperature  # adjust randomness

        probs = F.softmax(output, dim=-1)  # (1, vocab_size)
        next_token = torch.multinomial(probs, num_samples=1).item()
        generated.append(next_token)
        if next_token == 2 or next_token == 0: # reach end of sequence
          break

        input_token = torch.tensor([[next_token]], device=device)

    return generated

def note_frequency(midi_files):
    freqCount = {}
    for mFile in midi_files:
      pitch_list = note_extraction(mFile)
      for note in pitch_list:
        if note in freqCount:
          freqCount[note] += 1
        else:
          freqCount[note] = 1

    return freqCount

def note_unigram_probability(midi_files):
    note_counts = note_frequency(midi_files)
    unigramProbabilities = {}

    total = sum(note_counts.values())
    for key, value in note_counts.items():
      unigramProbabilities[key] = value/total

    return unigramProbabilities

def note_bigram_probability(midi_files):
    bigramTransitions = defaultdict(list)
    bigramTransitionProbabilities = defaultdict(list)

    #collect the bigrams
    bigrams = defaultdict(int)
    for mFile in midi_files:
      pitch_list = note_extraction(mFile)
      for (chord1,chord2) in zip(pitch_list[:-1],pitch_list[1:]):
        bigrams[(chord1,chord2)] += 1

    #use the bigrams (already unique and counted)
    transitionCounts = defaultdict(list)
    for key, value in bigrams.items():
      bigramTransitions[key[0]].append(key[1])
      transitionCounts[key[0]].append(value)

    #normalize counts into probabilities
    for key, value in transitionCounts.items():
      total = sum(value)
      for count in value:
        bigramTransitionProbabilities[key].append(count/total)

    return bigramTransitions, bigramTransitionProbabilities

def note_bigram_perplexity(midi_file):
    unigramProbabilities = note_unigram_probability(train_data)

    actual = note_extraction(midi_file)

    #prob get that sequence
    total = 0
    N = len(actual)
    for i in range(N):
      curr = actual[i]
      if i == 0:
        total += np.log(unigramProbabilities[curr]) #p(w1)
      else:
        prev = actual[i-1]
        for j in range(len(bigramTransitions[prev])):
          if bigramTransitions[prev][j] == curr: #aligning with probs list

        total += np.log(bigramTransitionProbabilities[prev][index])

    return (np.exp((-1/N)*total))

def extract_map(midi_file):
  midi_obj = miditoolkit.midi.parser.MidiFile(midi_file)
  ticks_per_beat = midi_obj.ticks_per_beat
  notes = midi_obj.instruments[0].notes
  max_time = max(n.end for n in notes)

  examples = []
  for t in range(0, max_time, ticks_per_beat):
      chord_notes = [n.pitch for n in notes if n.start <= t < n.end]
      # Get chord
      if len(chord_notes) < 3:
          continue

      # Get melody (highest pitch in this window)
      window_notes = [n for n in notes if t <= n.start < t + ticks_per_beat]
      if not window_notes:
          continue
      melody_note = max(window_notes, key=lambda n: n.pitch).pitch

      examples.append((chord_notes, melody_note))
  return examples

        self.data = []

        for i, file in enumerate(midi_files):

                examples = extract_map(file)  # returns [(chord_notes, melody_note)]
                for chord_notes, note in examples:

                # Save every N files
                if i % checkpoint_interval == 0:
                    print(f"Processed {i} files")
                    with open(save_path, "wb") as f:
                        pickle.dump(self.data, f)

            except Exception as e:
                print(f"Error processing {file}: {e}")

            pickle.dump(self.data, f)

class HarmonyRNN(nn.Module):
    def __init__(self, chord_vocab_size, embedding_dim, hidden_dim, num_layers, melody_vocab_size):
        super(HarmonyRNN, self).__init__()
        self.embedding = nn.Embedding(chord_vocab_size, embedding_dim)
        self.rnn = nn.LSTM(
            input_size=embedding_dim,
            hidden_size=hidden_dim,
            num_layers=num_layers,
            batch_first=True
        )
        self.fc1 = nn.Linear(hidden_dim, hidden_dim // 2)
        self.fc2 = nn.Linear(hidden_dim // 2, melody_vocab_size)

    def forward(self, x, hidden=None):
        x = self.embedding(x)
        out, hidden = self.rnn(x, hidden)
        out = F.relu(self.fc1(out))
        out = self.fc2(out)
        return out, hidden

def generate_melody(model, chord_sequence, temperature=1.0, device='cuda'):
    model.eval()
    chord_sequence = torch.tensor(chord_sequence, device=device).unsqueeze(0)  # (1, seq_len)

    with torch.no_grad():
        outputs, _ = model(chord_sequence)  # (1, seq_len, melody_vocab_size)
        logits = outputs.squeeze(0)  # (seq_len, melody_vocab_size)
        logits = logits / temperature
        probs = torch.softmax(logits, dim=-1)
        melody_tokens = torch.multinomial(probs, num_samples=1).squeeze(-1)  # (seq_len)

    return melody_tokens.cpu().tolist()

def create_midi(chord_seq, melody_seq, output_path):
    midi_obj = miditoolkit.MidiFile()
    piano = miditoolkit.Instrument(program=0)  # Acoustic Grand Piano

    # Example: align chords and melody notes per timestep (assuming fixed duration)
    time_per_step = 480 * 3
    time = 0

    for chord_pcs, melody_pitches in zip(chord_seq, melody_seq):
        # Add chord notes
        for pc in chord_pcs:
            note = miditoolkit.Note(
                velocity=80,
                pitch=pc,
                start=time,
                end=time + time_per_step
            )
            piano.notes.append(note)

        # Add melody notes consecutively within the chord step
        num_notes = len(melody_pitches)
        if num_notes > 0:
            note_duration = time_per_step // num_notes
            for i, pitch in enumerate(melody_pitches):
                if pitch >= 0:
                    note = miditoolkit.Note(
                        velocity=100,
                        pitch=pitch,
                        start=time + i * note_duration,
                        end=time + (i + 1) * note_duration
                    )
                    piano.notes.append(note)

        time += time_per_step

    midi_obj.instruments.append(piano)
    midi_obj.dump(output_path)

def evaluate_perplexity2(model, data_loader, melody_vocab_size=88, device='cuda'):
    model.eval()
    total_loss = 0.0
    total_tokens = 0
    criterion = nn.CrossEntropyLoss(reduction='sum')

    with torch.no_grad():
        for batch in data_loader:
            chords = batch['chords'].to(device)
            melodies = batch['melodies'].to(device)

            outputs, _ = model(chords)
            outputs = outputs.reshape(-1, melody_vocab_size)
            targets = melodies.reshape(-1)

            loss = criterion(outputs, targets)
            total_loss += loss.item()
            total_tokens += targets.size(0)

    avg_loss = total_loss / total_tokens
    perplexity = math.exp(avg_loss)
    return perplexity
